=== First Program Source Code ===
import cv2
import numpy as np
import matplotlib.pyplot as plt

# Read input image
img = cv2.imread("/content/images.jfif")

if img is None:
    print("Image not found")
else:
    # Convert image to RGB for displaying with matplotlib
    img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)

    # Convert image to grayscale for Sobel operator
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

    # Apply Sobel operator in X direction
    sobel_x = cv2.Sobel(gray, cv2.CV_64F, 1, 0, ksize=3)

    # Apply Sobel operator in Y direction
    sobel_y = cv2.Sobel(gray, cv2.CV_64F, 0, 1, ksize=3)

    # Convert gradients to uint8
    abs_x = cv2.convertScaleAbs(sobel_x)
    abs_y = cv2.convertScaleAbs(sobel_y)

    # Combine both gradients
    combined = cv2.addWeighted(abs_x, 0.5, abs_y, 0.5, 0)

    # Display results using matplotlib
    plt.figure(figsize=(12, 10))

    plt.subplot(2, 2, 1)
    plt.imshow(img_rgb)
    plt.title("Original Image")
    plt.axis("off")

    plt.subplot(2, 2, 2)
    plt.imshow(abs_x, cmap="gray")
    plt.title("Horizontal Gradient (Sobel X)")
    plt.axis("off")

    plt.subplot(2, 2, 3)
    plt.imshow(abs_y, cmap="gray")
    plt.title("Vertical Gradient (Sobel Y)")
    plt.axis("off")

    plt.subplot(2, 2, 4)
    plt.imshow(combined, cmap="gray")
    plt.title("Combined Edge Detection")
    plt.axis("off")

    plt.tight_layout()
    plt.show()

import cv2
import numpy as np
import matplotlib.pyplot as plt

# Read input image
img = cv2.imread("/content/images.jfif")

if img is None:
    print("Image not found")
else:
    # Convert image to RGB for displaying with matplotlib
    img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    rows, cols = img_rgb.shape[:2]

    # Define three original points
    pts1 = np.float32([
        [50, 50],
        [200, 50],
        [50, 200]
    ])

    # Define three destination points
    pts2 = np.float32([
        [100, 100],
        [250, 80],
        [80, 250]
    ])

    # Calculate affine transformation matrix
    matrix = cv2.getAffineTransform(pts1, pts2)

    print("Affine Transformation Matrix:")
    print(matrix)

    # Apply affine transformation
    result = cv2.warpAffine(img_rgb, matrix, (cols, rows))

    # Display original and transformed images using matplotlib
    plt.figure(figsize=(12, 6))

    plt.subplot(1, 2, 1)
    plt.imshow(img_rgb)
    plt.title("Original Image")
    plt.axis("off")

    plt.subplot(1, 2, 2)
    plt.imshow(result)
    plt.title("Affine Transformed Image")
    plt.axis("off")

    plt.tight_layout()
    plt.show()

=== Third Program Source Code ===
import cv2
import numpy as np
import matplotlib.pyplot as plt

# Read input image
img = cv2.imread("/content/images.jfif")

if img is None:
    print("Image not found")
else:
    # Convert BGR image to RGB for matplotlib and HSV for color filtering
    img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)

    # Define green color range in HSV space
    lower = np.array([35, 40, 40])
    upper = np.array([85, 255, 255])

    # Create binary mask
    mask = cv2.inRange(hsv, lower, upper)

    # Apply mask to original RGB image
    result = cv2.bitwise_and(img_rgb, img_rgb, mask=mask)

    # Display results using matplotlib
    plt.figure(figsize=(15, 5))

    plt.subplot(1, 3, 1)
    plt.imshow(img_rgb)
    plt.title("Original Image")
    plt.axis("off")

    plt.subplot(1, 3, 2)
    plt.imshow(mask, cmap="gray")
    plt.title("Binary Mask (Green Regions)")
    plt.axis("off")

    plt.subplot(1, 3, 3)
    plt.imshow(result)
    plt.title("Extracted Green Object")
    plt.axis("off")

    plt.tight_layout()
    plt.show()

import cv2
import base64
from IPython.display import HTML
import os

# Read input video
video_path = "/content/team 3.mp4"
cap = cv2.VideoCapture(video_path)

if not cap.isOpened():
    print(f"Cannot open video: {video_path}")
else:
    # Get video properties
    fps = cap.get(cv2.CAP_PROP_FPS)
    width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))

    frames = []

    # Read and store all frames
    while True:
        ret, frame = cap.read()
        if not ret:
            break
        frames.append(frame)

    cap.release()

    # Reverse the frames
    frames.reverse()

    # Create output video
    output_path = "reversed_video.mp4"
    if os.path.exists(output_path):
        os.remove(output_path)

    # Use standard mp4v codec
    fourcc = cv2.VideoWriter_fourcc(*'mp4v')
    out = cv2.VideoWriter(
        output_path,
        fourcc,
        fps,
        (width, height)
    )

    # Write reversed frames
    for frame in frames:
        out.write(frame)

    out.release()
    print(f"Video reversed successfully and saved to {output_path}!")
